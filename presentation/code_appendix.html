2. Module: Feature Engineering (ml/feature_engineering.py)

Transforms raw access logs into 43 ML-ready features (temporal, geo-velocity, rolling counts, device changes, behavioral z-scores).

"""
Feature engineering pipeline extracting temporal, geo-spatial, rolling-window, and device features.
"""
import numpy as np, pandas as pd
from math import radians, cos, sin, asin, sqrt

def haversine(lon1, lat1, lon2, lat2):
    if pd.isna(lon1) or pd.isna(lat1) or pd.isna(lon2) or pd.isna(lat2): return 0.0
    lon1, lat1, lon2, lat2 = map(radians, [float(lon1), float(lat1), float(lon2), float(lat2)])
    dlon, dlat = lon2 - lon1, lat2 - lat1
    a = sin(dlat / 2)**2 + cos(lat1) * cos(lat2) * sin(dlon / 2)**2
    return 6371.0 * (2 * asin(sqrt(min(1.0, a))))

class FeatureEngineer:
    def fit_transform(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()
        df["timestamp"] = pd.to_datetime(df["timestamp"])
        df.sort_values(["entity_id", "timestamp"], inplace=True)
        
        # Temporal
        df["hour_of_day"] = df["timestamp"].dt.hour
        df["is_off_hours"] = ((df["hour_of_day"] < 6) | (df["hour_of_day"] > 22)).astype(int)
        df["time_since_last_login"] = df.groupby("entity_id")["timestamp"].diff().dt.total_seconds().fillna(0)
        
        # Rolling counts & failure rates
        df["login_count_1h"] = self._rolling_count(df, window_sec=3600)
        df["login_count_6h"] = self._rolling_count(df, window_sec=21600)
        df["failed_auth_ratio_1h"] = self._rolling_failure_rate(df, window_sec=3600)
        
        # Geo-velocity (km/h)
        df["distance_km"] = df.apply(lambda r: haversine(r["lon"], r["lat"], r["prev_lon"], r["prev_lat"]), axis=1)
        df["geo_velocity_kmh"] = np.where(df["time_since_last_login"] > 0, df["distance_km"] / (df["time_since_last_login"] / 3600.0), 0.0).clip(upper=30000)
        
        # Device & Cold-start
        df["fingerprint_changed"] = (df["device_fingerprint"] != df.groupby("entity_id")["device_fingerprint"].shift(1)).astype(int)
        df["is_cold_start"] = (df.groupby("entity_id").cumcount() + 1 < 20).astype(int)
        
        return df.fillna(0)

3. Module: Detection Model (ml/detection_model.py)

XGBoost + LSTM Autoencoder ensemble scoring pipeline.

"""
Anomaly Detector combining XGBoost binary classifier (scale_pos_weight=50) and sequence LSTM.
Risk score = 0.6 * XGB_prob + 0.4 * Normalized_LSTM_error
"""
import numpy as np, xgboost as xgb

class AnomalyDetector:
    def __init__(self):
        self.xgb_clf = xgb.XGBClassifier(
            n_estimators=200, max_depth=6, learning_rate=0.1,
            scale_pos_weight=50, eval_metric="logloss", random_state=42
        )
        
    def predict(self, X_test, X_test_seq=None):
        xgb_prob = self.xgb_clf.predict_proba(X_test)[:, 1]
        risk_scores = np.clip(xgb_prob * 100, 0, 100)
        return risk_scores, (risk_scores > 50)
